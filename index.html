        document.getElementById('translation-text').style.display = 'none';

        document.getElementById('hint-text').textContent = "Cevabı Görmek İçin Tıkla (Space)";

        

        document.getElementById('main-show-btn').style.display = 'block';

        document.getElementById('anki-controls').style.display = 'none';

    }


    function speakWord(event) {

        if (event) event.stopPropagation();

        if (!currentCard || !currentCard.word) return;


        if ('speechSynthesis' in window) {

            window.speechSynthesis.cancel();

            const utterance = new SpeechSynthesisUtterance(currentCard.word);

            utterance.lang = 'en-US';

            utterance.rate = 0.85;

            window.speechSynthesis.speak(utterance);

        }

    }


    function showAnswer() {

        if (!currentCard || isFlipped) return;

        isFlipped = true;


        document.getElementById('card-icon').style.display = 'block';

        document.getElementById('meaning-text').style.display = 'block';

        if (currentCard.example) document.getElementById('example-text').style.display = 'block';

        if (currentCard.translation) document.getElementById('translation-text').style.display = 'block';

        document.getElementById('hint-text').textContent = "";


        document.getElementById('main-show-btn').style.display = 'none';

        document.getElementById('anki-controls').style.display = 'flex';

    }


    function processAnki(grade) {

        if (!currentCard || !isFlipped) return;


        let card = ankiData[currentCard.id];

        card.lastGrade = grade;


        const poppedCard = activeQueue.shift();


        if (grade === 1) {

            repeatQueue.push(poppedCard);

        }


        saveData();

        renderNextCard();

    }


    function deleteCardDirectly(cardId) {

        if (confirm("Bu kelimeyi listeden kalıcı olarak silmek istediğine emin misin?")) {

            delete ankiData[cardId];

            activeQueue = activeQueue.filter(c => c.id !== cardId);

            repeatQueue = repeatQueue.filter(c => c.id !== cardId);

            saveData();

            renderNextCard();

            updateFilterCounts();

            filterList('all', document.querySelector('.filter-chip.active'));

        }

    }


    function advanceToNextDay() {

        const password = prompt("Yeni güne geçmek için şifreyi girin:");

        if (password === "1213") {

            currentIndexPointer += DAILY_LIMIT;

            buildDailyQueue();

            saveData();

            renderNextCard();

            alert(`Başarılı! Yeni gün yüklendi. Sırada ${activeQueue.length} kelime var.`);

        } else if (password !== null) {

            alert("Hatalı şifre! İşlem iptal edildi.");

        }

    }


    function updateFilterCounts() {

        const counts = { all: 0, 0: 0, 1: 0, 2: 0, 3: 0, 4: 0 };

        Object.values(ankiData).forEach(item => {

            const g = item.lastGrade || 0;

            counts.all++;

            counts[g]++;

        });


        document.getElementById('filter-all').textContent = `Tümü (${counts.all})`;

        document.getElementById('filter-1').textContent = `🔴 Yeniden (${counts[1]})`;

        document.getElementById('filter-2').textContent = `🟠 Zor (${counts[2]})`;

        document.getElementById('filter-3').textContent = `🔵 İyi (${counts[3]})`;

        document.getElementById('filter-4').textContent = `🟢 Kolay (${counts[4]})`;

        document.getElementById('filter-0').textContent = `⚪ Bakılmadı (${counts[0]})`;

    }


    function openListModal() {

        document.getElementById('list-modal').style.display = 'flex';

        updateFilterCounts();

        filterList('all', document.getElementById('filter-all'));

    }


    function closeListModal() {

        document.getElementById('list-modal').style.display = 'none';

    }


    function filterList(filter, btn) {

        document.querySelectorAll('.filter-chip').forEach(b => b.classList.remove('active'));

        if (btn) btn.classList.add('active');


        const container = document.getElementById('modal-word-container');

        container.innerHTML = '';


        const labelMap = { 0: 'Bakılmadı', 1: 'Yeniden', 2: 'Zor', 3: 'İyi', 4: 'Kolay' };


        Object.values(ankiData).sort((a,b) => a.index - b.index).forEach(item => {

            const grade = item.lastGrade || 0;

            if (filter === 'all' || grade === filter) {

                const div = document.createElement('div');

                div.className = 'word-item';

                div.innerHTML = `

                    <div class="word-item-info">

                        <div class="word-item-title">${item.index + 1}. ${item.word}</div>

                        <div class="word-item-sub">${item.meaning}</div>

                    </div>

                    <div class="word-item-actions">

                        <span class="badge badge-${grade}">${labelMap[grade]}</span>

                        <button class="delete-btn" onclick="deleteCardDirectly('${item.id}')">Sil</button>

                    </div>

                `;

                container.appendChild(div);

            }

        });

    }


    document.addEventListener('keydown', function(e) {

        if (e.code === 'Space' || e.code === 'Enter') {

            e.preventDefault();

            if (!isFlipped) showAnswer();

        } else if (isFlipped) {

            if (e.key === '1') processAnki(1);

            if (e.key === '2') processAnki(2);

            if (e.key === '3') processAnki(3);

            if (e.key === '4') processAnki(4);

        }

    });
